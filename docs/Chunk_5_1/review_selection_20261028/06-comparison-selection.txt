=== ecas/ingestion/exomol_reference_cache.py ===
    1: """Atomic reference envelopes retain raw bytes, exact settings and provenance."""
    2: import base64
    3: from dataclasses import asdict, replace
    4: from datetime import datetime, timezone
    5: from hashlib import sha256
    6: import json
    7: import os
    8: import re
    9: from pathlib import Path
   10: import tempfile
   11: from uuid import uuid4
   12: 
   13: from .exomol_product import CITATIONS, ProductRequest, validate_product
   14: from .exomol_product_network import RetrievedCrossSection
   15: 
   16: MAX_CACHE_BYTES=2*1024*1024
   17: 
   18: 
   19: def _root(root):
   20:     root=Path(root)
   21:     if root.is_symlink() or root.is_junction(): raise ValueError('Linked reference cache is unsupported')
   22:     return root
   23: 
   24: 
   25: def _context(context, request):
   26:     if context is None: return
   27:     if not isinstance(context,dict) or set(context)!={'formula','isotopologue','dataset','version','catalogue_sha256'}:
   28:         raise ValueError('Invalid selected catalogue context')
   29:     service=request.service
   30:     if (context['formula'],context['isotopologue'],context['dataset'])!=(service.formula,service.isotopologue,service.dataset):
   31:         raise ValueError('Selected catalogue identity differs from supported service')
   32:     if not isinstance(context['version'],str) or not re.fullmatch('[0-9]{8}',context['version']):
   33:         raise ValueError('Invalid catalogue version label')
   34:     if not isinstance(context['catalogue_sha256'],str) or not re.fullmatch('[0-9a-f]{64}',context['catalogue_sha256']):
   35:         raise ValueError('Invalid catalogue source hash')
   36: 
   37: 
   38: def store_reference(root, result, *, catalogue_context=None):
   39:     # Recompute from raw bytes/settings; never trust a separately supplied parsed product.
   40:     checked=validate_product(result.raw,result.request)
   41:     if checked!=result.product: raise ValueError('Reference payload and parsed provenance disagree')
   42:     _context(catalogue_context, result.request)
   43:     envelope=dict(schema='ecas-exomol-reference-v1',request=asdict(result.request),
   44:         provenance=asdict(checked.provenance),retrieved_utc=datetime.now(timezone.utc).isoformat(),
   45:         source_sha256=checked.source_sha256,raw_base64=base64.b64encode(result.raw).decode('ascii'),
   46:         catalogue_context=catalogue_context)
   47:     unsigned=json.dumps(envelope,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
   48:     envelope['integrity_sha256']=sha256(unsigned).hexdigest()
   49:     encoded=json.dumps(envelope,allow_nan=False).encode()
   50:     if len(encoded)>MAX_CACHE_BYTES: raise ValueError('Reference cache record exceeds 2 MiB')
   51:     root=_root(root); root.mkdir(parents=True,exist_ok=True)
   52:     target=root/(uuid4().hex+'.json')
   53:     temporary=None
   54:     try:
   55:         with tempfile.NamedTemporaryFile(dir=root,prefix='reference-',suffix='.tmp',delete=False) as stream:
   56:             temporary=Path(stream.name); stream.write(encoded); stream.flush(); os.fsync(stream.fileno())
   57:         os.replace(temporary,target)
   58:     finally:
   59:         if temporary is not None: temporary.unlink(missing_ok=True)
   60:     return target
   61: 
   62: 
   63: def load_reference(path):
   64:     path=Path(path)
   65:     _root(path.parent)
   66:     if path.is_symlink() or path.is_junction(): raise ValueError('Linked reference record is unsupported')
   67:     with path.open('rb') as stream: raw=stream.read(MAX_CACHE_BYTES+1)
   68:     if len(raw)>MAX_CACHE_BYTES: raise ValueError('Reference cache record exceeds 2 MiB')
   69:     try:
   70:         envelope=json.loads(raw)
   71:         required={'schema','request','provenance','retrieved_utc','source_sha256','raw_base64','integrity_sha256'}
   72:         if set(envelope) not in (required,required|{'catalogue_context'}):
   73:             raise ValueError('Unexpected reference envelope fields')
   74:         unsigned={k:v for k,v in envelope.items() if k!='integrity_sha256'}
   75:         if sha256(json.dumps(unsigned,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()!=envelope['integrity_sha256']:
   76:             raise ValueError('Reference record integrity mismatch')
   77:         if envelope['schema']!='ecas-exomol-reference-v1': raise ValueError('Unsupported reference record')
   78:         stamp=datetime.fromisoformat(envelope['retrieved_utc'])
   79:         if stamp.utcoffset() is None or stamp.utcoffset().total_seconds()!=0: raise ValueError('UTC timestamp required')
   80:         request=ProductRequest(**envelope['request'])
   81:         _context(envelope.get('catalogue_context'), request)
   82:         source=base64.b64decode(envelope['raw_base64'],validate=True)
   83:         product=validate_product(source,request)
   84:         # JSON round-trip normalizes tuple citations to lists without altering values.
   85:         expected=json.loads(json.dumps(asdict(product.provenance)))
   86:         stored=envelope['provenance']
   87:         if request.isotopologue=='12C-16O2' and isinstance(stored,dict) and stored.get('citations')==list(CITATIONS[:2]):
   88:             # Early v1 records retained the method/dataset citations before the
   89:             # general ExoMol citation was added. Preserve, rather than rewrite,
   90:             # those valid historical metadata while validating every other field.
   91:             expected['citations']=list(CITATIONS[:2])
   92:             product=replace(product,provenance=replace(product.provenance,citations=CITATIONS[:2]))
   93:         if expected!=stored or product.source_sha256!=envelope['source_sha256']:
   94:             raise ValueError('Saved reference provenance differs from validated request')
   95:         return RetrievedCrossSection(request,source,product)
   96:     except (ValueError,TypeError,KeyError,RecursionError) as error:
   97:         raise ValueError(f'Invalid reference cache: {error}') from error
   98: 
   99: 
  100: def cached_references(root):
  101:     root=_root(root)
  102:     if not root.exists(): return [],0
  103:     records=[]; invalid=0
  104:     for path in sorted(root.glob('*.json')):
  105:         try: records.append((path,load_reference(path)))
  106:         except (OSError,ValueError): invalid+=1
  107:     return records,invalid
