77: def matching_search_url(url, name):
78:     """Compare the generated query semantically while preserving original evidence.
79:
80:     NASA returns canonical target casing even for case-insensitive searches.
81:     Decode query parameters before comparing so Unicode/percent encoding works;
82:     keep the endpoint and parameter structure exact, including duplicate rejection.
83:     """
84:     if not isinstance(url, str):
85:         return False
86:     try:
87:         actual, expected = urlsplit(url), urlsplit(spectra_search(name))
88:         # Keep empty values so blank duplicates/unknown parameters cannot vanish
89:         # during validation. Reject malformed query components explicitly.
90:         params = parse_qs(actual.query, keep_blank_values=True, strict_parsing=True)
91:         wanted = parse_qs(expected.query, keep_blank_values=True, strict_parsing=True)
92:     except ValueError:
93:         return False
94:     if actual._replace(query="") != expected._replace(query=""):
95:         return False
96:     return (set(params) == set(wanted) and params.get('format') == wanted['format']
97:             and len(params.get('query', [])) == 1
98:             and params['query'][0].casefold() == wanted['query'][0].casefold())

101: def spectrum_url(page, record):
102:     """Follow NASA's inspected BaseURL + /data/ + SPEC_PATH construction.
103:
104:     Workspace comes from the current page, not a stale saved session. No remote
105:     scripts are executed. If NASA changes this supported layout, stop explicitly.
106:     """
107:     if record["spec_type"] != "Transmission" or not valid_spec_path(record["spec_path"]):
108:         raise ValueError("Only supported transmission records can be downloaded")
109:     if type(page) is not bytes or len(page) > MAX_BYTES:
110:         raise ValueError("Invalid NASA interface page")
111:     text = page.decode("utf-8")
112:     matches = re.findall(r"FF_InitPage\s*\(\s*'[^']*'\s*,\s*'(/workspace/TMP_[A-Za-z0-9_]+)'", text)
113:     if len(matches) != 1:
114:         raise ValueError("NASA session layout changed; cannot resolve download safely")
115:     return HOST + matches[0] + "/atmospheres/tab1/data/" + record["spec_path"]
