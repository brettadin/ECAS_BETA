   53:     def load_spectrum(self, spectrum_data, schema=None) -> bool:
   54:         """Plot valid coordinate pairs from a normalized SpectrumData object.
   55: 
   56:         Returns True if at least one valid point is plotted. Returns False
   57:         and clears the widget if the required columns or valid points are absent.
   58:         The 'schema' argument is used to override column name detection for
   59:         archive schemas like WASP-39 b, which use non-standard column names.
   60: 
   61:         Args:
   62:             spectrum_data: The SpectrumData object containing normalized data.
   63:             schema: Optional schema dict mapping ECAS canonical names to source columns.
   64:                     If None, generic column name detection is used.
   65:         """
   66: 
   67:         # Clear the previous plot before attempting a replacement. A failed
   68:         # load must never leave the previous dataset visible.
   69:         self.clear()
   70: 
   71:         if spectrum_data is None:
   72:             return False
   73: 
   74:         # Determine wavelength and signal columns based on schema.
   75:         available_columns = set(spectrum_data.columns)
   76: 
   77:         if schema is not None and len(schema) == 2:
   78:             # Archive schema provides direct mapping from ECAS canonical names.
   79:             self._wavelength_column = schema.get("wavelength")
   80:             self._signal_column = schema.get("signal")
   81:         else:
   82:             # Generic mode uses simple heuristic for common column names.
   83:             self._wavelength_column = (
   84:                 "wavelength" if "wavelength" in available_columns else None
   85:             )
   86:             self._signal_column = (
   87:                 "signal" if "signal" in available_columns else None
   88:             )
   89: 
   90:         # Reject data without both required coordinate columns.
   91:         if not (self._wavelength_column and self._signal_column):
   92:             self.reset_on_failure()
   93:             return False
   94: 
   95:         # Use only units explicitly supplied by SpectrumData.
   96:         self._axis_unit_label = spectrum_data.axis_unit
   97:         self._signal_unit_label = spectrum_data.signal_unit
   98: 
   99:         # Filter missing, non-numeric, NaN, and infinite coordinate pairs.
  100:         x_vals = []
  101:         y_vals = []
  102: 
  103:         for row in spectrum_data.rows:
  104:             wavelength = row.get(self._wavelength_column)
  105:             signal = row.get(self._signal_column)
  106: 
  107:             if wavelength is None or signal is None:
  108:                 continue
  109: 
  110:             try:
  111:                 x_value = float(wavelength)
  112:                 y_value = float(signal)
  113:             except (TypeError, ValueError, OverflowError):
  114:                 continue
  115: 
  116:             if not (math.isfinite(x_value) and math.isfinite(y_value)):
  117:                 continue
  118: 
  119:             x_vals.append(x_value)
  120:             y_vals.append(y_value)
  121: 
  122:         # An empty plot must not retain metadata from a failed dataset.
  123:         if not x_vals:
  124:             self.reset_on_failure()
  125:             return False
  126: 
  127:         # Plot the actual valid measurements; no interpolation is performed.
  128:         scatter = ScatterPlotItem(
  129:             x=x_vals,
  130:             y=y_vals,
  131:             pen=None,
  132:             brush=qtgui.QBrush(qtgui.QColor("steelblue")),
  133:             symbol="o",
  134:             symbolSize=6,
  135:             antialias=True,
  136:         )
  137:         self.plot_widget.addItem(scatter)
  138: 
  139:         # Show units only when they are known.
  140:         xlabel = self._wavelength_column
  141:         ylabel = self._signal_column
  142: 
  143:         if self._axis_unit_label:
  144:             xlabel = f"{xlabel} ({self._axis_unit_label})"
  145: 
  146:         if self._signal_unit_label:
  147:             ylabel = f"{ylabel} ({self._signal_unit_label})"
  148: 
  149:         self.plot_widget.setLabel("bottom", text=xlabel)
  150:         self.plot_widget.setLabel("left", text=ylabel)
  151: 
  152:         # Let PyQtGraph scale the axes to the supplied valid measurements.
  153:         # Padding also handles datasets with identical minimum and maximum.
  154:         self.viewbox.setRange(
  155:             xRange=(min(x_vals), max(x_vals)),
  156:             yRange=(min(y_vals), max(y_vals)),
  157:             padding=0.05,
  158:         )
  159: 
  160:         # Keep the original model object; never modify its rows or metadata.
  161:         self._current_spectrum = spectrum_data
  162:         self.dataset_changed.emit(spectrum_data)
  163:         return True
  164:
