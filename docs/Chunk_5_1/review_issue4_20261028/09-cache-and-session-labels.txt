   73: def _load_reference(path):

   75:     _root(path.parent)
   76:     if path.is_symlink() or path.is_junction(): raise ValueError('Linked reference record is unsupported')
   77:     with path.open('rb') as stream: raw=stream.read(MAX_CACHE_BYTES+1)
   78:     if len(raw)>MAX_CACHE_BYTES: raise ValueError('Reference cache record exceeds 2 MiB')
   79:     try:
   80:         envelope=json.loads(raw)
   81:         required={'schema','request','provenance','retrieved_utc','source_sha256','raw_base64','integrity_sha256'}
   82:         if set(envelope) not in (required,required|{'catalogue_context'}):
   83:             raise ValueError('Unexpected reference envelope fields')
   84:         unsigned={k:v for k,v in envelope.items() if k!='integrity_sha256'}
   85:         if sha256(json.dumps(unsigned,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()!=envelope['integrity_sha256']:
   86:             raise ValueError('Reference record integrity mismatch')
   87:         if envelope['schema']!='ecas-exomol-reference-v1': raise ValueError('Unsupported reference record')
   88:         stamp=datetime.fromisoformat(envelope['retrieved_utc'])
   89:         if stamp.utcoffset() is None or stamp.utcoffset().total_seconds()!=0: raise ValueError('UTC timestamp required')
   90:         request=ProductRequest(**envelope['request'])
   91:         _context(envelope.get('catalogue_context'), request)
   92:         source=base64.b64decode(envelope['raw_base64'],validate=True)
   93:         product=validate_product(source,request)
   94:         # JSON round-trip normalizes tuple citations to lists without altering values.
   95:         expected=json.loads(json.dumps(asdict(product.provenance)))
   96:         stored=envelope['provenance']
