    2: """Typed representation of an imported spectrum.

   10: 
   11: 
   12: # CSV cells can be numeric, text, or missing.
   13: # None represents a missing value; it must never be treated as zero.
   14: CellValue = float | str | None
   15: 
   16: 
   17: @dataclass(frozen=True)
   18: class SpectrumData:
   19:     """Structured data returned after a spectrum file is imported."""
   20: 
   21:     # Original file location, retained for traceability.
   22:     source_file: Path
   23: 
   24:     # Column names and validated rows, in their original column order.
   25:     rows: tuple[dict[str, CellValue], ...]
   26:     columns: tuple[str, ...]
   27: 
   28:     # None means the importer has not established this information.
   29:     # These values must not be inferred from the data alone.
   30:     axis_unit: str | None = None
   31:     signal_unit: str | None = None
   32:     signal_type: str | None = None
   33: 
   34:     # Non-fatal import limitations, such as missing uncertainty values.
   35:     warnings: tuple[str, ...] = ()
   36: 
   37:     @property
   38:     def row_count(self) -> int:
   39:         """Return the number of imported measurement rows."""
   40:         return len(self.rows)
