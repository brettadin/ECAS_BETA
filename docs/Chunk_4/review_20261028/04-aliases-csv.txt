65:         if self.reply is not reply or self.failure:
66:             return
67:         while reply.bytesAvailable():
68:             self.buffer.extend(bytes(reply.read(min(65536, MAX_BYTES + 1 - len(self.buffer)))))
69:             if len(self.buffer) > MAX_BYTES:
70:                 self.cancel("NASA response exceeds 1 MiB")
71:                 return
72: 
73:     def _finished(self, reply):
74:         if self.reply is not reply:
75:             return
76:         self._read(reply)
77:         # A final drain can overflow and abort. abort() may synchronously emit
78:         # finished again, whose callback can already have started a new request.
79:         if self.reply is not reply:
80:             return
81:         self.timer.stop()
82:         error = self.failure
83:         status = reply.attribute(QNetworkRequest.Attribute.HttpStatusCodeAttribute)
84:         if not error and (reply.error() != QNetworkReply.NetworkError.NoError or status != 200):
85:             error = f"NASA request failed (HTTP {status}): {reply.errorString()}"
86:         callback, raw = self.callback, bytes(self.buffer)
87:         self.reply = None
88:         self.callback = None
89:         self.buffer.clear()
90:         reply.deleteLater()
